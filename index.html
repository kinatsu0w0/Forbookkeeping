function doPost(e) {
  var sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getActiveSheet();
  var data = JSON.parse(
    e.postData.contents
  );
  data.transactions.forEach(function(r) {
    sheet.appendRow([
      r.date, r.transactionType,
      r.mainCategory, r.subCategory,
      r.description, r.amount,
      r.paymentMethod, r.note, r.tag
    ]);
  });
  return ContentService
    .createTextOutput('OK');
}
